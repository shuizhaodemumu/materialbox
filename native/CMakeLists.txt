cmake_minimum_required(VERSION 3.22)
project(materialbox_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(sherpa_c SHARED IMPORTED)
set_target_properties(sherpa_c PROPERTIES IMPORTED_LOCATION "${SHERPA_LIBS}/libsherpa-onnx-c-api.so")
add_library(materialbox_speech SHARED speech.cpp)
target_include_directories(materialbox_speech PRIVATE "${SHERPA_INCLUDE}")
target_link_libraries(materialbox_speech PRIVATE sherpa_c log)
target_link_options(materialbox_speech PRIVATE "-Wl,-z,max-page-size=16384")
